**Tech Stack:** TypeScript (CharCell/TerminalPanel pre-text rendering), Python/FastAPI, Bonfires KG (VISITED_BY edges).

```python
@router.get("/worldmap/{player_id}")
async def get_player_world_map(player_id: str):
    """Get fog-of-war world map — only visited rooms + adjacent unknowns."""
    try:
        from memento.bonfires_client import get_client
        client = await asyncio.to_thread(get_client)

        # Get player's visited locations via VISITED_BY edges
        visited_edges = await asyncio.to_thread(
            client.kg.get_edges, player_id, direction="outgoing", edge_type="VISITED_BY"
        )
        visited_ids: set[str] = set()
        for edge in visited_edges:
            target_id = edge.get("target_node_uuid", edge.get("target_uuid", ""))
            if target_id:
                visited_ids.add(target_id)

        # Get current location
        from gateway.app import ws_hub
        current_location = ws_hub.player_locations.get(player_id, "") if ws_hub else ""

        # Build rooms list from visited locations
        rooms = []
        all_connections = []

        for loc_id in visited_ids:
            try:
                entity = await asyncio.to_thread(client.kg.get_entity, loc_id)
                if isinstance(entity, dict) and "entity" in entity:
                    entity = entity["entity"]
                name = entity.get("name", "Unknown")
                rooms.append({"id": loc_id, "name": name, "visited": True})

                # Get exits from room_map
                import json
                attrs = entity.get("attributes", {})
                if isinstance(attrs, str):
                    attrs = json.loads(attrs) if attrs else {}
                summary = entity.get("summary", "")
                # Try to get room_map from summary (it's stored as JSON in summary)
                rm = {}
                if isinstance(summary, str) and summary.startswith("{"):
                    try:
                        parsed = json.loads(summary)
                        rm = parsed if "exits" in parsed else {}
                    except (json.JSONDecodeError, TypeError):
                        pass

                for exit_info in rm.get("exits", []):
                    target = exit_info.get("target", "")
                    direction = exit_info.get("direction", "")
                    target_id = exit_info.get("target_id", "")
                    if direction:
                        all_connections.append({
                            "from_id": loc_id,
                            "from_name": name,
                            "to_id": target_id,
                            "to_name": target,
                            "direction": direction,
                        })
            except Exception:
                continue

        # Add unknown rooms (adjacent to visited but not visited themselves)
        for conn in all_connections:
            to_id = conn["to_id"]
            if to_id and to_id not in visited_ids:
                # Unknown room — show as "?" with direction info
                rooms.append({
                    "id": to_id,
                    "name": "?",
                    "visited": False,
                    "direction": conn["direction"],
                    "from_id": conn["from_id"],
                })

        return {
            "current": current_location,
            "rooms": rooms,
            "connections": [
                {"from": c["from_name"], "to": c["to_name"], "direction": c["direction"]}
                for c in all_connections
            ],
        }
    except Exception:
        import logging
        logging.getLogger(__name__).error("Player world map failed", exc_info=True)
        return {"current": "", "rooms": [], "connections": []}
```

### Task 4: Compass Rose Panel Renderer

**Files:**

- [ ] **Step 1: Create the worldmap panel renderer**

```typescript
// client/src/panels/worldmap.ts
/**
 * World map panel — ASCII compass rose showing current room,
 * visited neighbors, and undiscovered exits. Pre-text rendered
 * on TerminalPanel canvas. Clickable directions trigger navigation.
 */

import type { WorldMapData, WorldMapConnection } from '../state/game-state';
import type { TerminalPanel } from '../ui/terminal-panel';
import type { CharCell } from '../renderer/canvas-text';
import { theme } from '../renderer/theme';
import { textRow, coloredRow, emptyRow } from './panel-utils';

/** Direction → row/col offset in the compass grid. */
const DIR_OFFSETS: Record<string, [number, number]> = {
  north: [0, -1],
  south: [0, 1],
  east: [1, 0],
  west: [-1, 0],
  up: [0, -1],
  down: [0, 1],
  northeast: [1, -1],
  northwest: [-1, -1],
  southeast: [1, 1],
  southwest: [-1, 1],
};

export function renderWorldMapPanel(
  panel: TerminalPanel,
  worldMap: WorldMapData | null,
  currentLocation: string,
  onAction: (action: string) => void,
): void {
  const cols = panel.cols;
  const cells: CharCell[][] = [];
  panel.clearHitRegions();

  if (!worldMap || !currentLocation) {
    cells.push(textRow('No map data', theme.colors.dim, cols));
    panel.paint(cells);
    return;
  }

  // Find connections from current room
  const currentExits = worldMap.connections.filter(
    c => c.from.toLowerCase() === currentLocation.toLowerCase()
  );

  // Build compass: center = current, directional neighbors around it
  const neighbors: Record<string, { name: string; visited: boolean; direction: string }> = {};
  for (const conn of currentExits) {
    const dir = conn.direction.toLowerCase();
    const room = worldMap.rooms.find(
      r => r.name.toLowerCase() === conn.to.toLowerCase()
    );
    neighbors[dir] = {
      name: room?.name || conn.to || '?',
      visited: room?.visited ?? false,
      direction: dir,
    };
  }

  // Also add unvisited rooms that reference current room
  for (const room of worldMap.rooms) {
    if (!room.visited && room.direction) {
      const fromRoom = worldMap.rooms.find(r => r.id === room.from_id);
      if (fromRoom && fromRoom.name.toLowerCase() === currentLocation.toLowerCase()) {
        const dir = room.direction.toLowerCase();
        if (!neighbors[dir]) {
          neighbors[dir] = { name: '?', visited: false, direction: dir };
        }
      }
    }
  }

  // Render compass rose as CharCell rows
  const currentColor = '#8b5cf6';
  const visitedColor = theme.colors.location || '#7aa2d4';
  const unknownColor = theme.colors.dim || '#3a3a48';
  const lineColor = theme.colors.dim || '#3a3a48';

  // Current room label (shortened to fit)
  const currentLabel = currentLocation.length > cols - 4
    ? currentLocation.slice(0, cols - 7) + '...'
    : currentLocation;

  // Helper to build a room label row centered
  function centeredRoomRow(name: string, color: string): CharCell[] {
    const text = name;
    const pad = Math.max(0, Math.floor((cols - text.length) / 2));
    const row: CharCell[] = [];
    for (let i = 0; i < cols; i++) {
      const ci = i - pad;
      if (ci >= 0 && ci < text.length) {
        row.push({ char: text[ci], fg: color });
      } else {
        row.push({ char: ' ', fg: lineColor });
      }
    }
    return row;
  }

  function centeredLineRow(ch: string): CharCell[] {
    const mid = Math.floor(cols / 2);
    const row: CharCell[] = [];
    for (let i = 0; i < cols; i++) {
      row.push({ char: i === mid ? ch : ' ', fg: lineColor });
    }
    return row;
  }

  // Row tracking for hit regions
  let rowIdx = 0;

  // North neighbor
  const north = neighbors['north'] || neighbors['up'];
  if (north) {
    cells.push(centeredRoomRow(north.name, north.visited ? visitedColor : unknownColor));
    panel.registerHitRegion({ col: 0, row: rowIdx, width: cols, height: 1, data: { action: `go ${north.direction}` } });
    rowIdx++;
    cells.push(centeredLineRow('|'));
    rowIdx++;
  }

  // Center row: west — [current] — east
  const west = neighbors['west'];
  const east = neighbors['east'];
  const westLabel = west ? (west.visited ? west.name : '?') : '';
  const eastLabel = east ? (east.visited ? east.name : '?') : '';
  const centerText = `[${currentLabel}]`;

  // Build center row with west, current, east
  const centerRow: CharCell[] = [];
  const totalWidth = cols;
  const centerStart = Math.floor((totalWidth - centerText.length) / 2);

  // West section
  const westEnd = centerStart - 2; // space for " -- "
  const westStart = Math.max(0, westEnd - westLabel.length);
  const westColor = west ? (west.visited ? visitedColor : unknownColor) : lineColor;

  for (let i = 0; i < totalWidth; i++) {
    if (west && i >= westStart && i < westStart + westLabel.length) {
      centerRow.push({ char: westLabel[i - westStart], fg: westColor });
    } else if (west && i >= westStart + westLabel.length && i < centerStart) {
      centerRow.push({ char: i === westStart + westLabel.length + 1 ? '-' : (i > westStart + westLabel.length + 1 && i < centerStart ? '-' : ' '), fg: lineColor });
    } else if (i >= centerStart && i < centerStart + centerText.length) {
      centerRow.push({ char: centerText[i - centerStart], fg: currentColor });
    } else if (east && i === centerStart + centerText.length + 1) {
      centerRow.push({ char: '-', fg: lineColor });
    } else if (east && i > centerStart + centerText.length + 1 && i < centerStart + centerText.length + 3) {
      centerRow.push({ char: '-', fg: lineColor });
    } else if (east && i >= centerStart + centerText.length + 3 && i < centerStart + centerText.length + 3 + eastLabel.length) {
      centerRow.push({ char: eastLabel[i - (centerStart + centerText.length + 3)], fg: east.visited ? visitedColor : unknownColor });
    } else {
      centerRow.push({ char: ' ', fg: lineColor });
    }
  }

  cells.push(centerRow);
  // Register hit regions for west and east
  if (west) {
    panel.registerHitRegion({ col: westStart, row: rowIdx, width: westLabel.length, height: 1, data: { action: `go ${west.direction}` } });
  }
  if (east) {
    const eastStart = centerStart + centerText.length + 3;
    panel.registerHitRegion({ col: eastStart, row: rowIdx, width: eastLabel.length, height: 1, data: { action: `go ${east.direction}` } });
  }
  rowIdx++;

  // South neighbor
  const south = neighbors['south'] || neighbors['down'];
  if (south) {
    cells.push(centeredLineRow('|'));
    rowIdx++;
    cells.push(centeredRoomRow(south.name, south.visited ? visitedColor : unknownColor));
    panel.registerHitRegion({ col: 0, row: rowIdx, width: cols, height: 1, data: { action: `go ${south.direction}` } });
    rowIdx++;
  }

  // Fill remaining space if panel is taller
  while (cells.length < 3) {
    cells.push(emptyRow(cols));
  }

  panel.paint(cells);
}
```
