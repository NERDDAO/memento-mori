```ts
    case "room_draw": {
      if (msg.kind !== "revealed") break; // deepened/exhausted: no new glyph
      const entity = msg.entity as { uuid?: string; name?: string } | null;
      if (!entity || !entity.name) break;
      const things = s.viewport.currentThings();
      if (things.some((t) => t.uuid === entity.uuid || t.name === entity.name)) break;
      s.viewport.setContents(s.roomUuid(), [

      ]);
      break;
    }
```

```ts
// The fixed opening epigraph (a title card shown BEFORE /opening/start, so it
// cannot come from the start response). Mirrors the engine's OPENING_EPIGRAPH.
export const OPENING_EPIGRAPH =
  "The old world is dying, yet the new world struggles to be born: this is the time of monsters.";

/**
 * Show the title card (Memento Mori + epigraph fade), reveal the name box, and
 * call `onName(name)` once on submit (Enter key or button). Idempotent-safe:
 * fires `onName` at most once, then hides the overlay.
 */
export function mountOpeningIntro(onName: (name: string) => void): void {
  const overlay = document.getElementById("opening-intro");
  const epigraph = document.getElementById("opening-intro-epigraph");
  const namebox = document.getElementById("opening-intro-namebox");
  const input = document.getElementById("opening-intro-name") as HTMLInputElement | null;
  const enter = document.getElementById("opening-intro-enter");
  if (!overlay || !epigraph || !namebox || !input || !enter) {
    onName("wanderer"); // intro markup missing — degrade to the default identity
    return;
  }
  epigraph.textContent = OPENING_EPIGRAPH;
  // reveal the name box after the epigraph has faded in
  window.setTimeout(() => {
    namebox.removeAttribute("hidden");
    input.focus();
  }, 3200);

  let done = false;
  const submit = (): void => {
    if (done) return;
    done = true;
    const name = (input.value.trim() || "wanderer").slice(0, 30);
    overlay.classList.add("hidden");
    onName(name);
  };
  enter.addEventListener("click", submit);
  input.addEventListener("keydown", (e) => {
    if ((e as KeyboardEvent).key === "Enter") submit();
  });
}
```
